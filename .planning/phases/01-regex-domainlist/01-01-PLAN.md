type regexServeMux struct {
    mu       sync.RWMutex
    exact    map[string]dns.HandlerFunc
    wildcard []struct {
        prefix  string
        handler dns.HandlerFunc
    }
    regex    []regexHandler
    catchAll dns.HandlerFunc
}

type regexHandler struct {
    pattern *regexp.Regexp
    handler func(dns.ResponseWriter, *dns.Msg)
}

Constructor (exact):

func newRegexServeMux() *regexServeMux {
    return &regexServeMux{
        exact:    make(map[string]dns.HandlerFunc),
        wildcard: make([]struct{ prefix string; handler dns.HandlerFunc }, 0),
        regex:    make([]regexHandler, 0),
    }
}

Methods — implement in this order:

1. HandleFunc(pattern string, handler dns.HandlerFunc)
   - m.mu.Lock(); defer m.mu.Unlock()
   - if strings.HasPrefix(pattern, "*."): m.wildcard = append(m.wildcard, struct{prefix string; handler dns.HandlerFunc}{prefix: pattern[2:], handler: handler}); return
   - m.exact[dns.CanonicalName(pattern)] = handler

2. HandleRegex(pattern string, handler dns.HandlerFunc) error
   - re, err := regexp.Compile(pattern)
   - if err != nil: return fmt.Errorf("invalid regex %q: %w", pattern, err)
   - m.mu.Lock(); defer m.mu.Unlock()
   - m.regex = append(m.regex, regexHandler{pattern: re, handler: handler})
   - return nil

3. HandleRemove(pattern string)
   - m.mu.Lock(); defer m.mu.Unlock()
   - if strings.HasPrefix(pattern, "*."):
     prefix := pattern[2:]  // FIX: strip "*." (2 chars) to match HandleFunc
     for i, wc := range m.wildcard {
       if wc.prefix == prefix {
         m.wildcard = append(m.wildcard[:i], m.wildcard[i+1:]...)
         return
       }
     }
     return
   - delete(m.exact, pattern)

4. SetCatchAll(handler dns.HandlerFunc)
   - m.mu.Lock(); defer m.mu.Unlock()
   - m.catchAll = handler

5. clear()
   - m.mu.Lock(); defer m.mu.Unlock()
   - m.exact = make(map[string]dns.HandlerFunc)
   - m.wildcard = m.wildcard[:0]
   - m.regex = m.regex[:0]
   - m.catchAll = nil

6. ServeDNS(w dns.ResponseWriter, r *dns.Msg)
   - if len(r.Question) == 0: if m.catchAll != nil { m.catchAll(w, r) }; return
   - question := r.Question[0].Name
   - if !strings.HasSuffix(question, "."): question = question + "."
   - m.mu.RLock(); defer m.mu.RUnlock()
   - canonical := strings.TrimSuffix(question, ".")
   - if handler, ok := m.exact[canonical]; ok { handler(w, r); return }
   - for _, wc := range m.wildcard { if strings.HasSuffix(canonical, wc.prefix+".") { wc.handler(w, r); return } }
   - for _, rh := range m.regex { if rh.pattern.MatchString(canonical) { rh.handler(w, r); return } }
   - if m.catchAll != nil { m.catchAll(w, r) }

CRITICAL constraints from RESEARCH.md:
- Pitfall #1: Do NOT call dns.HandleFunc or dns.HandleRemove — these go to the global DefaultServeMux which is never used
- Pitfall #2: Use dns.CanonicalName() for registration keys (adds trailing dot); use strings.TrimSuffix(question, ".") for ServeDNS lookups
- Pitfall #3: Shutdown must clear custom mux handlers (not call dns.HandleRemove)
- Pitfall #4: Use sync.RWMutex — RLock in ServeDNS, Lock in registration methods
- Anti-pattern: Do NOT embed dns.ServeMux — use fully standalone implementation (spike validated)
- Anti-pattern: Do NOT use regexp.MustCompile() — use regexp.Compile() with error handling per D-04

WARNING 1 Fix: HandleRemove wildcard prefix strip changed from pattern[1:] to pattern[2:] (strip "*." not just "*"). This ensures consistency with HandleFunc which also strips 2 characters for wildcard detection.

## Threat Model

### Trust Boundaries

| Boundary | Description |
|----------|-------------|
| domainlist file → regexServeMux | Untrusted regex patterns enter system from operator-controlled file |
| DNS query → ServeDNS | External DNS clients send queries that traverse priority routing |

### STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | E (Elevation) | regexServeMux.HandleRegex | mitigate | Patterns validated at load time via regexp.Compile() — invalid patterns return error, never compiled at runtime |
| T-01-02 | A (Availability) | regexServeMux.ServeDNS | accept | ReDoS risk mitigated by trusting operator-controlled domainlist file; daemon runs as root on controlled network per D-03 |
| T-01-03 | Tampering | regexServeMux registration | mitigate | sync.RWMutex protects concurrent access to handler maps; clear() uses write lock to atomically reset all handlers |
| T-01-SC | Tampering | npm/pip/cargo installs | mitigate | slopcheck + blocking human checkpoint for [ASSUMED]/[SUS] |
