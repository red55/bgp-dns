       f, e := os.Open(fn)

       }

           if err != nil {
               log.L().Warn().Msgf("Failed to close file: %v", err)

       // SAFE RELOAD: build tempMux first, swap only on success
       // This prevents handler loss if file scan fails mid-way
       tempMux := newRegexServeMux()
       oldGen := c.generation()

       scanner := bufio.NewScanner(f)
       for scanner.Scan() {
           line := strings.TrimSpace(scanner.Text())
           if len(line) == 0 { continue }
           if line[0] == '#' || line[0] == ';' { continue }

           if strings.HasPrefix(line, "regex:") {
               pattern := strings.TrimPrefix(line, "regex:")
               if err := c.registerRegexOn(line, tempMux); err != nil {
                   c.L().Warn().Msgf("Skipping invalid regex pattern %q: %v", pattern, err)
                   continue
               }
           } else {
               if e = c.registerOn(line, tempMux, false); e != nil {
                   return e
               }
           }
       }

       // All registrations succeeded — atomically swap mux
       c.mux = tempMux
       _ = c.increaseGeneration()

       return c.evictByGeneration(oldGen)
   }

## Threat Model

### Trust Boundaries

| Boundary | Description |
|----------|-------------|
| domainlist file → cache.load() → c.mux.HandleRegex() | Untrusted regex patterns enter system from operator-controlled file |
| DNS query → _server.Handler (mux) → ServeDNS | External DNS clients send queries that traverse priority routing |

### STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | E (Elevation) | cache.load() → c.mux.HandleRegex() | mitigate | Patterns validated at load time via regexp.Compile() — invalid patterns return error, logged as Warn, skipped (per D-04) |
| T-01-02 | A (Availability) | ServeDNS priority routing | accept | ReDoS risk mitigated by trusting operator-controlled domainlist file; daemon runs as root on controlled network per D-03 |
| T-01-03 | Tampering | cache.register()/unregister() | mitigate | All handler registration goes through custom mux with sync.RWMutex protection (no global mux pollution) |
| T-01-04 | Elevation | cache.load() safe reload | mitigate | BLOCKER 3 fix: load() builds tempMux first, atomically swaps only on success — prevents handler loss if file scan fails mid-way |
| T-01-SC | Tampering | npm/pip/cargo installs | mitigate | slopcheck + blocking human checkpoint for [ASSUMED]/[SUS] |
